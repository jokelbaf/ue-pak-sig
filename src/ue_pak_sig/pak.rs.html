5const PAK_MAGIC: u32 = 0x5A6F12E1;
6
7/// Minimal pak file footer, containing only the fields needed for sig
8/// generation and verification.
9#[derive(Debug, Clone)]
10pub struct PakFooter {
11    /// Pak format version.
12    pub version: i32,
13    /// Byte offset of the primary index within the pak file.
14    pub index_offset: i64,
15    /// Byte length of the primary index.
16    pub index_size: i64,
17    /// SHA1 hash of the raw primary index data, as stored in the footer.
18    pub index_hash: [u8; 20],
19    /// Whether the primary index is AES-encrypted.
20    pub encrypted_index: bool,
21}
22
23impl PakFooter {
24    /// Read and parse the pak footer from a seekable reader.
25    ///
26    /// Tries a set of known footer sizes to handle multiple pak format
27    /// versions (UE 4.20–4.27).
28    pub fn read<R: Read + Seek>(reader: &mut R) -> Result<Self, Error> {
29        // Candidate footer sizes (bytes from end of file):
30        //   v8+ (FNameBasedCompressionMethod): 16+1+4+4+8+8+20+160 = 221
31        //   v9  (FrozenIndex, extra bool):      221+1              = 222
32        //   v7  (EncryptionKeyGuid, no methods): 16+1+4+4+8+8+20  =  61
33        //   v3-6 (no guid, no methods):          1+4+4+8+8+20     =  45
34        const CANDIDATES: &[usize] = &[221, 222, 61, 45];
35
36        let file_len = reader.seek(SeekFrom::End(0))?;
37
38        for &footer_size in CANDIDATES {
39            if file_len < footer_size as u64 {
40                continue;
41            }
42
43            reader.seek(SeekFrom::End(-(footer_size as i64)))?;
44            let mut buf = vec![0u8; footer_size];
45            reader.read_exact(&mut buf)?;
46
47            if let Some(footer) = try_parse_footer(&buf) {
48                return Ok(footer);
49            }
50        }
51
52        Err(Error::InvalidPak("pak footer magic not found".into()))
53    }
54}
55
56fn try_parse_footer(buf: &[u8]) -> Option<PakFooter> {
57    for i in 0..buf.len().saturating_sub(3) {
58        let magic = u32::from_le_bytes(buf[i..i + 4].try_into().ok()?);
59        if magic != PAK_MAGIC {
60            continue;
61        }
62
63        let remaining = buf.len() - i;
64        if remaining < 4 + 4 + 8 + 8 + 20 {
65            continue;
66        }
67
68        let version = i32::from_le_bytes(buf[i + 4..i + 8].try_into().ok()?);
69        let index_offset = i64::from_le_bytes(buf[i + 8..i + 16].try_into().ok()?);
70        let index_size = i64::from_le_bytes(buf[i + 16..i + 24].try_into().ok()?);
71        let mut index_hash = [0u8; 20];
72        index_hash.copy_from_slice(&buf[i + 24..i + 44]);
73
74        // bEncryptedIndex should be at buf[i - 1], one byte before magic
75        let encrypted_index = i > 0 && buf[i - 1] != 0;
76
77        return Some(PakFooter {
78            version,
79            index_offset,
80            index_size,
81            index_hash,
82            encrypted_index,
83        });
84    }
85    None
86}

88#[cfg(test)]
89mod tests {
90    use super::*;
91    use std::io::Cursor;
92
93    fn make_footer_buf(
94        version: i32,
95        index_offset: i64,
96        index_size: i64,
97        hash: [u8; 20],
98    ) -> Vec<u8> {
99        let mut buf = Vec::new();
100        // EncryptionKeyGuid (16 bytes)
101        buf.extend_from_slice(&[0u8; 16]);
102        // bEncryptedIndex (1 byte)
103        buf.push(0);
104        // Magic
105        buf.extend_from_slice(&PAK_MAGIC.to_le_bytes());
106        // Version
107        buf.extend_from_slice(&version.to_le_bytes());
108        // IndexOffset
109        buf.extend_from_slice(&index_offset.to_le_bytes());
110        // IndexSize
111        buf.extend_from_slice(&index_size.to_le_bytes());
112        // IndexHash
113        buf.extend_from_slice(&hash);
114        // CompressionMethods (160 bytes)
115        buf.extend_from_slice(&[0u8; 160]);
116        buf
117    }
118
119    #[test]
120    fn round_trip_footer() {
121        let hash = [0xABu8; 20];
122        let buf = make_footer_buf(11, 12345, 6789, hash);
123        let footer = PakFooter::read(&mut Cursor::new(&buf)).unwrap();
124        assert_eq!(footer.version, 11);
125        assert_eq!(footer.index_offset, 12345);
126        assert_eq!(footer.index_size, 6789);
127        assert_eq!(footer.index_hash, hash);
128    }
