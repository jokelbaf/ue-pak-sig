14pub(crate) const SIG_MAGIC: u32 = 0x73832DAA;
15pub(crate) const SIG_VERSION: i32 = 1;
16
17/// Unreal Engine version variant that determines the signing scheme.
18///
19/// The on-disk `.sig` format is identical for both versions; they differ
20/// only in what data is RSA-signed.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum UeVersion {
23    /// UE 4.26 — signs `SHA1(chunk_crcs)` (20 bytes).
24    V426,
25    /// UE 4.27 — signs `IndexHash || SHA1(chunk_crcs)` (40 bytes),
26    /// where `IndexHash` is the SHA1 of the pak's primary index as stored
27    /// in the pak footer.
28    V427,
29}
30
31/// A parsed UE4 `.sig` file.
32#[derive(Debug, Clone)]
33pub struct SigFile {
34    /// RSA-encrypted block produced by `FRSA::EncryptPrivate`.
35    pub encrypted_hash: Vec<u8>,
36    /// CRC32 hashes of each contiguous 64 KiB chunk of the corresponding
37    /// `.pak` file, in order.
38    pub chunk_hashes: Vec<u32>,
39}
40
41impl SigFile {
42    /// Deserialize a `.sig` file from `reader`.
43    pub fn read<R: Read>(reader: &mut R) -> Result<Self, Error> {
44        let magic = read_u32(reader)?;
45        if magic != SIG_MAGIC {
46            return Err(Error::InvalidMagic(magic));
47        }
48
49        let version = read_i32(reader)?;
50        if version != SIG_VERSION {
51            return Err(Error::UnsupportedVersion(version));
52        }
53
54        let enc_len = read_i32(reader)? as usize;
55        let mut encrypted_hash = vec![0u8; enc_len];
56        reader.read_exact(&mut encrypted_hash)?;
57
58        let chunk_count = read_i32(reader)? as usize;
59        let mut chunk_hashes = Vec::with_capacity(chunk_count);
60        for _ in 0..chunk_count {
61            chunk_hashes.push(read_u32(reader)?);
62        }
63
64        Ok(Self {
65            encrypted_hash,
66            chunk_hashes,
67        })
68    }
69
70    /// Deserialize a `.sig` file from `path`.
71    pub fn read_from_file(path: &Path) -> Result<Self, Error> {
72        let mut f = File::open(path)?;
73        Self::read(&mut f)
74    }
75
76    /// Serialize this `.sig` file to `writer`.
77    pub fn write<W: Write>(&self, writer: &mut W) -> Result<(), Error> {
78        writer.write_all(&SIG_MAGIC.to_le_bytes())?;
79        writer.write_all(&SIG_VERSION.to_le_bytes())?;
80
81        writer.write_all(&(self.encrypted_hash.len() as i32).to_le_bytes())?;
82        writer.write_all(&self.encrypted_hash)?;
83
84        writer.write_all(&(self.chunk_hashes.len() as i32).to_le_bytes())?;
85        for &h in &self.chunk_hashes {
86            writer.write_all(&h.to_le_bytes())?;
87        }
88
89        Ok(())
90    }
91
92    /// Serialize this `.sig` file to `path`.
93    pub fn write_to_file(&self, path: &Path) -> Result<(), Error> {
94        let mut f = File::create(path)?;
95        self.write(&mut f)
96    }
97
98    /// Compute `SHA1(chunk_hashes)` — the "master hash" that is embedded
99    /// inside the RSA-encrypted block.
100    pub fn master_hash(&self) -> [u8; 20] {
101        crypto::sha1_of_u32_slice(&self.chunk_hashes)
102    }
103
104    /// Sign a `.pak` file and produce a `SigFile`.
105    ///
106    /// Reads the pak file in streaming 64 KiB chunks to avoid loading the
107    /// entire file into memory.  For `UeVersion::V427`, also reads the pak
108    /// footer to extract the primary index hash.
109    pub fn sign(pak_path: &Path, key: &SigningKey, version: UeVersion) -> Result<Self, Error> {
110        let f = File::open(pak_path)?;
111        let mut reader = BufReader::new(f);
112        Self::sign_from_reader(&mut reader, key, version)
113    }
114
115    /// Sign from an arbitrary `Read + Seek` source.
116    pub fn sign_from_reader<R: Read + Seek>(
117        reader: &mut R,
118        key: &SigningKey,
119        version: UeVersion,
120    ) -> Result<Self, Error> {
121        let chunk_hashes = compute_chunk_hashes(reader)?;
122        let master = crypto::sha1_of_u32_slice(&chunk_hashes);
123
124        let plaintext = build_sign_plaintext(reader, version, &master)?;
125        let encrypted_hash = key.sign(&plaintext)?;
126
127        Ok(Self {
128            encrypted_hash,
129            chunk_hashes,
130        })
131    }
132
133    /// Verify this `.sig` file against a `.pak` file.
134    ///
135    /// Checks that:
136    /// 1. The chunk CRC32s match those computed from the pak file.
137    /// 2. The RSA signature decrypts to the expected plaintext.
138    pub fn verify(
139        &self,
140        pak_path: &Path,
141        key: &SigningKey,
142        version: UeVersion,
143    ) -> Result<(), Error> {
144        let f = File::open(pak_path)?;
145        let mut reader = BufReader::new(f);
146        self.verify_from_reader(&mut reader, key, version)
147    }
148
149    /// Verify against an arbitrary `Read + Seek` source.
150    pub fn verify_from_reader<R: Read + Seek>(
151        &self,
152        reader: &mut R,
153        key: &SigningKey,
154        version: UeVersion,
155    ) -> Result<(), Error> {
156        let computed = compute_chunk_hashes(reader)?;
157        if computed != self.chunk_hashes {
158            return Err(Error::SignatureMismatch);
159        }
160
161        let master = crypto::sha1_of_u32_slice(&self.chunk_hashes);
162        let plaintext = build_sign_plaintext(reader, version, &master)?;
163        key.verify(&plaintext, &self.encrypted_hash)
164    }
165}
166
167fn compute_chunk_hashes<R: Read + Seek>(reader: &mut R) -> Result<Vec<u32>, Error> {
168    reader.seek(SeekFrom::Start(0))?;
169    let mut hashes = Vec::new();
170    let mut buf = vec![0u8; CHUNK_SIZE];
171
172    loop {
173        let n = read_chunk(reader, &mut buf)?;
174        if n == 0 {
175            break;
176        }
177        hashes.push(crypto::crc32(&buf[..n]));
178    }
179
180    Ok(hashes)
181}
182
183fn read_chunk<R: Read>(reader: &mut R, buf: &mut [u8]) -> Result<usize, Error> {
184    let mut total = 0;
185    while total < buf.len() {
186        match reader.read(&mut buf[total..]) {
187            Ok(0) => break,
188            Ok(n) => total += n,
189            Err(e) if e.kind() == std::io::ErrorKind::Interrupted => {}
190            Err(e) => return Err(e.into()),
191        }
192    }
193    Ok(total)
194}
195
196fn build_sign_plaintext<R: Read + Seek>(
197    reader: &mut R,
198    version: UeVersion,
199    master_hash: &[u8; 20],
200) -> Result<Vec<u8>, Error> {
201    match version {
202        UeVersion::V426 => Ok(master_hash.to_vec()),
203        UeVersion::V427 => {
204            let footer = PakFooter::read(reader)?;
205            let mut plaintext = Vec::with_capacity(40);
206            plaintext.extend_from_slice(&footer.index_hash);
207            plaintext.extend_from_slice(master_hash);
208            Ok(plaintext)
209        }
210    }
211}

213fn read_u32<R: Read>(r: &mut R) -> Result<u32, Error> {
214    let mut b = [0u8; 4];
215    r.read_exact(&mut b)?;
216    Ok(u32::from_le_bytes(b))
217}
218
219fn read_i32<R: Read>(r: &mut R) -> Result<i32, Error> {
220    let mut b = [0u8; 4];
221    r.read_exact(&mut b)?;
222    Ok(i32::from_le_bytes(b))
223}

225#[cfg(test)]
226mod tests {
227    use super::*;

229    #[test]
230    fn round_trip_serialization() {
231        let sig = SigFile {
232            encrypted_hash: vec![0xAAu8; 256],
233            chunk_hashes: vec![0x11223344, 0xDEADBEEF],
234        };
235
236        let mut buf = Vec::new();
237        sig.write(&mut buf).unwrap();
238
239        let parsed = SigFile::read(&mut buf.as_slice()).unwrap();
240        assert_eq!(parsed.encrypted_hash, sig.encrypted_hash);
241        assert_eq!(parsed.chunk_hashes, sig.chunk_hashes);
242    }
243
244    #[test]
245    fn wrong_magic_returns_error() {
246        let mut buf = 0u32.to_le_bytes().to_vec();
247        buf.extend_from_slice(&1i32.to_le_bytes());
248        let result = SigFile::read(&mut buf.as_slice());
249        assert!(matches!(result, Err(Error::InvalidMagic(_))));
250    }
251
252    #[test]
253    fn master_hash_is_sha1_of_le_u32s() {
254        let sig = SigFile {
255            encrypted_hash: vec![],
256            chunk_hashes: vec![1u32, 2u32],
257        };
258        let expected = crypto::sha1_of_u32_slice(&[1, 2]);
259        assert_eq!(sig.master_hash(), expected);
260    }
