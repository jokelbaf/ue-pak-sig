1//! # ue-pak-sig
2//!
3//! Generate and verify Unreal Engine 4.26/4.27 `.sig` files.
4//!
5//! ## Quick start
6//!
7//! ```no_run
8//! use ue_pak_sig::{CryptoConfig, SigningKey, SigFile, UeVersion};
9//! use std::path::Path;
10//!
11//! let config = CryptoConfig::from_file(Path::new("config.ini")).unwrap();
12//! let key = SigningKey::from_config(&config).unwrap();
13//!
14//! // Sign a pak file.
15//! let sig = SigFile::sign(Path::new("MyMod.pak"), &key, UeVersion::V427).unwrap();
16//! sig.write_to_file(Path::new("MyMod.sig")).unwrap();
17//!
18//! // Verify a sig file.
19//! let sig = SigFile::read_from_file(Path::new("MyMod.sig")).unwrap();
20//! sig.verify(Path::new("MyMod.pak"), &key, UeVersion::V427).unwrap();
21//! ```
