1use rsa::rand_core::OsRng;
2use rsa::{BigUint, RsaPrivateKey, RsaPublicKey, pkcs1v15::Pkcs1v15Sign, traits::SignatureScheme};
3
4use crate::{error::Error, ini::CryptoConfig};
5
6/// RSA signing key pair loaded from a `CryptoConfig`.
7///
8/// Holds the public key always; the private key is present only when the
9/// config includes `SigningPrivateExponent` (i.e. editor/developer configs).
10pub struct SigningKey {
11    public: RsaPublicKey,
12    private: Option<RsaPrivateKey>,
13}
14
15impl SigningKey {
16    /// Construct a `SigningKey` from a parsed `CryptoConfig`.
17    ///
18    /// Returns an error if the config is missing the modulus or public
19    /// exponent, or if the RSA components are invalid.
20    pub fn from_config(config: &CryptoConfig) -> Result<Self, Error> {
21        let modulus = config
22            .signing_modulus
23            .as_deref()
24            .ok_or_else(|| Error::InvalidIni("missing SigningModulus".into()))?;
25
26        let pub_exp = config
27            .signing_public_exponent
28            .as_deref()
29            .ok_or_else(|| Error::InvalidIni("missing SigningPublicExponent".into()))?;
30
31        // UE stores RSA components as little-endian big integers.
32        let n = BigUint::from_bytes_le(modulus);
33        let e = BigUint::from_bytes_le(pub_exp);
34
35        let public =
36            RsaPublicKey::new(n.clone(), e.clone()).map_err(|e| Error::Rsa(e.to_string()))?;
37
38        let private = config
39            .signing_private_exponent
40            .as_deref()
41            .map(|priv_exp| {
42                let d = BigUint::from_bytes_le(priv_exp);
43                RsaPrivateKey::from_components(n, e, d, vec![])
44                    .map_err(|e| Error::Rsa(e.to_string()))
45            })
46            .transpose()?;
47
48        Ok(Self { public, private })
49    }
50
51    /// Returns `true` if a private key is available for signing.
52    pub fn has_private_key(&self) -> bool {
53        self.private.is_some()
54    }
55
56    /// Sign `data` using RSA private key with PKCS#1 v1.5 Type 1 padding
57    /// (no DigestInfo prefix), matching UE's `FRSA::EncryptPrivate`.
58    pub fn sign(&self, data: &[u8]) -> Result<Vec<u8>, Error> {
59        let priv_key = self
60            .private
61            .as_ref()
62            .ok_or_else(|| Error::Rsa("private key not available".into()))?;
63
64        Pkcs1v15Sign::new_unprefixed()
65            .sign(Some(&mut OsRng), priv_key, data)
66            .map_err(|e| Error::Rsa(e.to_string()))
67    }
68
69    /// Verify a signature against `expected_plaintext` using the public key,
70    /// matching UE's `FRSA::DecryptPublic`.
71    pub fn verify(&self, expected_plaintext: &[u8], signature: &[u8]) -> Result<(), Error> {
72        Pkcs1v15Sign::new_unprefixed()
73            .verify(&self.public, expected_plaintext, signature)
74            .map_err(|_| Error::SignatureMismatch)
75    }
76}
